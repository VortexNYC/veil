```ts
import { createAmbientClient, listItems, useItem } from "@vortex-api/veil";

// Zero static keys in env vars: resolves Vercel OIDC (x-vercel-oidc-token /
// VERCEL_OIDC_TOKEN_FILE), GitHub Actions OIDC, or VEIL_OIDC_TOKEN.
const client = createAmbientClient();

const { data } = await listItems({ client });
await useItem({
  client,
  body: {
    item: "stripe",
    url: "https://api.stripe.com/v1/customers",
    method: "POST",
  },
});
```

`listItems` is metadata. `useItem` injects. The vault secret is not in the response.
